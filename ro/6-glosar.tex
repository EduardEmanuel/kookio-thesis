}

\newglossaryentry{analog}{

\newglossaryentry{angular}{

\newglossaryentry{nitro}{

\newglossaryentry{zod}{

  name={\NoHyphens{PostgreSQL}},

}


\newglossaryentry{vercel}{
  name={\NoHyphens{Vercel}},

\newglossaryentry{playwright}{

\newglossaryentry{hydration}{

\newglossaryentry{workflow}{
  name={\NoHyphens{workflow}},
  sort={workflow},
  plural={\NoHyphens{workflow-uri}},
  description={configurație declarativă care descrie o succesiune de pași automatizați (build, testare, lint, deploy) declanșați de evenimente (ex.: push, pull request, programare). În ecosisteme precum GitHub Actions și GitLab \glsxtrshort{ci}, un workflow este compus din job-uri și pași, poate folosi matrici, cache, artefacte, secrete/variabile și reguli de concurență. În acest proiect, workflow-urile orchestrează \glsxtrshort{cicd} și aliniază execuția la țintele \gls{nx}.},
  see={[vezi și]{cicd,nx}},
}

\newglossaryentry{compositeaction}{
  name={\NoHyphens{composite action}},
  sort={compositeaction},
  plural={\NoHyphens{composite actions}},
  description={tip de acțiune reutilizabilă în GitHub Actions care grupează mai mulți pași (\texttt{run}/\texttt{uses}) într-o singură acțiune; se distribuie ca director cu fișier \texttt{action.yml}, acceptă \textit{inputs}, \textit{outputs} și \textit{secrets} și poate fi apelată dintr-un \gls{workflow} prin \texttt{uses:}. Utilă pentru a elimina duplicarea (DRY) în pipeline-urile \glsxtrshort{cicd}; limitări uzuale: nu rulează în containere proprii și nu pot defini \textit{services}.},
  see={[vezi și]{workflow,cicd}},
}


\newglossaryentry{redis}{
  name={\NoHyphens{Redis}},
  sort={redis},
  description={magazin de date \emph{in-memory} tip cheie–valoare, cu structuri avansate (liste, seturi, hash-uri, mulțimi ordonate, \textit{streams}) și latență foarte mică; folosit frecvent pentru caching, stocare sesiuni, cozi/mesagerie (Pub/Sub, Streams), rate-limiting și \textit{leaderboards}. Suportă persistență opțională (RDB/AOF), replicare, \textit{cluster} pentru partiționare și, de regulă, completează o bază SQL principală în arhitecturi microservicii și \textit{serverless}.},
}


\newglossaryentry{metaframework}{
  name={\NoHyphens{meta-framework}},
  sort={metaframework},
  description={framework care extinde un framework de bază pentru capabilități full-stack (ex. rutare pe fișiere, \glsxtrshort{ssr}/\glsxtrshort{ssg}, integrare cu \glsxtrshort{trpc}). În această lucrare, \gls{analog} extinde \gls{angular} peste runtime-ul \gls{nitro}.},
  see={[vezi și]{analog,angular,nitro,ssr,ssg,trpc}},
}


\newglossaryentry{randare}{
  name={\NoHyphens{randare}},
  sort={randare},
  description={procesul prin care o aplicație generează HTML (și actualizări ale interfeței) din cod și date. În web-ul modern poate avea loc pe server (\glsxtrshort{ssr}), la build-time prin \glsxtrshort{ssg}/prerender, sau în browser (\glsxtrshort{csr}); de regulă, este urmată de \gls{hydration} pentru interactivitate. În \gls{analog}, strategia de randare poate fi aleasă per rută.},
  see={[vezi și]{ssr,ssg,csr,hydration,analog}},
}


\newglossaryentry{logging}{
  name={\NoHyphens{jurnalizare (logging)}},
  sort={logging},
  description={colectarea și înregistrarea evenimentelor aplicației (mesaje structurate pe niveluri, corelare prin identificatori, export către sisteme de agregare) pentru depanare și observabilitate},
}

\newglossaryentry{prismaclient}{
  name={\NoHyphens{Prisma Client}},
  sort={prismaclient},
  description={client \texttt{TypeScript} pentru baza de date, generat automat de \gls{prisma} din fișierul \texttt{schema.prisma}; oferă un API tipizat pentru operații SQL (CRUD, filtre, relații, tranzacții) asupra modelelor definite. Începând cu \gls{prisma} 7, clientul rulează integral în JavaScript (fără motor nativ în Rust): conexiunea trece printr-un \textit{driver adapter} (\texttt{@prisma/adapter-pg}), iar regenerarea cu \texttt{npx prisma generate} produce doar cod tipizat, independent de platforma de execuție.},
  see={[vezi și]{prisma,orm,cockroachdb,postgresql}},
}
